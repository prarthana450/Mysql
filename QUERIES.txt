CREATE DATABASE company;
USE company;
CREATE TABLE Employee (
    EmpNo INT PRIMARY KEY AUTO_INCREMENT,
    EName VARCHAR(50),
    Job VARCHAR(50),
    Sal DECIMAL(10,2),
    Comm DECIMAL(10,2),
    Mgr INT
);
